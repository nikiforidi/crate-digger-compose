events {
    worker_connections 1024;
}

http {
    upstream api_gateway {
        server gateway:8000;
    }

    upstream minio_backend {
        server minio:9000;
    }

    # Rate Limiting
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=100r/s;
    limit_req_zone $binary_remote_addr zone=minio_limit:10m rate=500r/s;

    # Cache zone for MinIO images
    proxy_cache_path /var/cache/nginx/minio levels=1:2
                     keys_zone=minio_cache:10m
                     max_size=1g
                     inactive=24h
                     use_temp_path=off;

    # ═══════════════════════════════════════════════════
    # DEV: HTTP server (порт 80)
    # ═══════════════════════════════════════════════════
    server {
        listen 80;
        server_name localhost;

        client_max_body_size 20M;

        # ── MinIO проксирование ──────────────────────
        location /minio/ {
            limit_req zone=minio_limit burst=100 nodelay;

            proxy_pass http://minio_backend/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Скрываем внутренние заголовки MinIO
            proxy_hide_header x-amz-request-id;
            proxy_hide_header x-amz-id-2;
            proxy_hide_header x-amz-version-id;

            # CORS для изображений
            add_header Access-Control-Allow-Origin * always;
            add_header Access-Control-Allow-Methods "GET, HEAD, OPTIONS" always;
            add_header Access-Control-Allow-Headers "Range, Origin, Content-Type, Accept" always;

            # Кэширование статики
            add_header Cache-Control "public, max-age=31536000, immutable" always;
            add_header X-Cache-Status $upstream_cache_status always;

            # Кэш
            proxy_cache minio_cache;
            proxy_cache_valid 200 24h;
            proxy_cache_valid 404 1m;
            proxy_cache_use_stale error timeout updating http_500 http_502 http_503 http_504;
            proxy_cache_key "$scheme$request_method$host$request_uri";

            if ($request_method = 'OPTIONS') {
                add_header Access-Control-Allow-Origin *;
                add_header Access-Control-Allow-Methods "GET, HEAD, OPTIONS";
                add_header Access-Control-Max-Age 1728000;
                add_header Content-Type "text/plain charset=UTF-8";
                add_header Content-Length 0;
                return 204;
            }
        }

        # ── API Gateway ──────────────────────────────
        location /api/ {
            limit_req zone=api_limit burst=20 nodelay;

            proxy_pass http://api_gateway;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_read_timeout 120s;
        }

        # ── Frontend (Next.js) ───────────────────────
        location / {
            proxy_pass http://marketplace-frontend:3000;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }

        # ── Health check ─────────────────────────────
        location /health {
            access_log off;
            return 200 '{"status":"ok","service":"nginx"}';
            add_header Content-Type application/json;
        }
    }

    # ═══════════════════════════════════════════════════
